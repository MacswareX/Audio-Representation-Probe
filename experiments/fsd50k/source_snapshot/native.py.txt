class EnCodec:
    sample_rate=24000;reps=('encodec_z','encodec_q2','encodec_q4')
    def __init__(self):
        from transformers import EncodecModel
        from safetensors.torch import load_file
        self.model,info=EncodecModel.from_pretrained(str(CORE/'downloads'),local_files_only=True,use_safetensors=True,output_loading_info=True)
        assert not any(info.get(k) for k in ('missing_keys','unexpected_keys','mismatched_keys','error_msgs'))
        self.model.load_state_dict(load_file(str(CORE/'downloads/model.safetensors')),strict=True);self.model.eval().requires_grad_(False);self.before=model_hash(self.model)
    @torch.inference_mode()
    def extract(self,x,check=False):
        capture={}
        def hook(m,args,out):capture['z']=out.detach()
        h=self.model.encoder.register_forward_hook(hook)
        try:e=self.model.encode(x[None,None],padding_mask=torch.ones_like(x[None,None],dtype=torch.bool),bandwidth=3.,return_dict=True)
        finally:h.remove()
        t=(len(x)+319)//320;codes=e.audio_codes;z=capture['z'];assert codes.shape==(1,1,4,t) and z.shape==(1,128,t)
        cb=codes[0].transpose(0,1);values={'encodec_z':(z[0].numpy(),np.arange(t)*320/24000)};accum=torch.zeros_like(z);errors=[]
        for k in range(4):
            accum=accum+F.embedding(cb[k],self.model.quantizer.layers[k].codebook.embed).permute(0,2,1)
            if k in (1,3):
                q=self.model.quantizer.decode(cb[:k+1]);error=float(abs(q-accum).max());assert torch.allclose(q,accum,atol=1e-6,rtol=1e-5);errors.append(error)
                values['encodec_q'+str(k+1)]=(q[0].numpy(),np.arange(t)*320/24000)
        if check:
            q15=self.model.encode(x[None,None],padding_mask=torch.ones_like(x[None,None],dtype=torch.bool),bandwidth=1.5,return_dict=True)
            assert torch.equal(q15.audio_codes,codes[:,:,:2])
        return values,dict(input_samples=len(x),native_frames=t,active_quantizers=4,total_quantizers=32,prefix_lookup_max_abs=errors,native15_equal=bool(check),context='native state reset for every call',extra_audio_padding=0)
