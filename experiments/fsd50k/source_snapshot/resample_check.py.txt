start();status('CORE_RESAMPLE_CHECK');results=[]

    with wave.open(row['raw_path'],'rb') as f:x=torch.from_numpy(np.frombuffer(f.readframes(f.getnframes()),dtype='<i2').astype(np.float32)/32768.)

    del x,a,b
